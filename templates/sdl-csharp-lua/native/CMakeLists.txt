cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0168)
    cmake_policy(SET CMP0168 NEW)
endif()
project(__PROJECT_IDENTIFIER___native LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type for single-config generators" FORCE)
endif()
if("__SDL_BACKEND__" STREQUAL "direct3d11" AND NOT WIN32)
    message(FATAL_ERROR "The Direct3D 11 starter requires Windows.")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "Build SDL statically")
set(SDL_STATIC ON CACHE BOOL "Build SDL statically")
set(SDL_TESTS OFF CACHE BOOL "Build SDL tests")
set(SDL_TEST_LIBRARY OFF CACHE BOOL "Build the SDL test library")
set(SDL_INSTALL OFF CACHE BOOL "Install SDL development files")
FetchContent_Declare(sdl3
    URL https://github.com/libsdl-org/SDL/releases/download/release-__SDL_VERSION__/SDL3-__SDL_VERSION__.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sdl3)
include(cmake/Lua.cmake)

add_library(pw_native SHARED src/scene.cpp)
target_compile_features(pw_native PRIVATE cxx_std_17)
target_compile_definitions(pw_native PRIVATE PW_SDL_BACKEND="__SDL_BACKEND__")
target_link_libraries(pw_native PRIVATE SDL3::SDL3-static lua_runtime)
set_target_properties(pw_native PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_BINARY_DIR}/bin>"
    LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_BINARY_DIR}/bin>")
if(WIN32)
    set_target_properties(pw_native PROPERTIES PREFIX "")
endif()

install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses/sdl3 COMPONENT AppRuntime)
install(FILES "${lua_SOURCE_DIR}/doc/readme.html" DESTINATION licenses/lua COMPONENT AppRuntime)

if("__SDL_BACKEND__" STREQUAL "vulkan")
    if(WIN32)
        set(PROJECT_VULKAN_RUNTIME_DIR "$ENV{PROJECT_VULKAN_RUNTIME_DIR}" CACHE PATH "Custom Vulkan redistributable folder")
        set(vulkan_arch x64)
        if("${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" MATCHES "ARM64")
            set(vulkan_arch arm64)
        elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
            set(vulkan_arch x86)
        endif()
        configure_file(cmake/VulkanRuntime.cmake.in VulkanRuntime.cmake @ONLY)
        install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/VulkanRuntime.cmake" COMPONENT AppRuntime)
    elseif(APPLE)
        find_file(PROJECT_MOLTENVK_LIBRARY NAMES libMoltenVK.dylib HINTS "$ENV{VULKAN_SDK}/lib")
        if(NOT PROJECT_MOLTENVK_LIBRARY)
            message(STATUS "Set VULKAN_SDK or PROJECT_MOLTENVK_LIBRARY to package MoltenVK.")
        endif()
        install(CODE "if(NOT EXISTS \"${PROJECT_MOLTENVK_LIBRARY}\")\n  message(FATAL_ERROR \"Install MoltenVK or set PROJECT_MOLTENVK_LIBRARY before publishing.\")\nendif()" COMPONENT AppRuntime)
        if(PROJECT_MOLTENVK_LIBRARY)
            file(REAL_PATH "${PROJECT_MOLTENVK_LIBRARY}" moltenvk)
            install(FILES "${moltenvk}" DESTINATION . RENAME libMoltenVK.dylib COMPONENT AppRuntime)
        endif()
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_file(PROJECT_VULKAN_LOADER NAMES libvulkan.so.1
            PATHS /usr/lib /lib PATH_SUFFIXES "${CMAKE_LIBRARY_ARCHITECTURE}" lib64)
        install(CODE "if(NOT EXISTS \"${PROJECT_VULKAN_LOADER}\")\n  message(FATAL_ERROR \"Install the Vulkan loader before publishing.\")\nendif()" COMPONENT AppRuntime)
        if(PROJECT_VULKAN_LOADER)
            file(REAL_PATH "${PROJECT_VULKAN_LOADER}" loader)
            install(FILES "${loader}" DESTINATION lib RENAME libvulkan.so.1 COMPONENT AppRuntime)
        endif()
    endif()
endif()
