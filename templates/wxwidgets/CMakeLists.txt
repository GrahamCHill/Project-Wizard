cmake_minimum_required(VERSION 3.24)
project(__PROJECT_IDENTIFIER__ LANGUAGES CXX)

add_executable(app WIN32 MACOSX_BUNDLE src/main.cpp)
target_compile_features(app PRIVATE cxx_std_17)

option(PROJECT_USE_SYSTEM_WXWIDGETS "Use an installed wxWidgets SDK when available" ON)
if(PROJECT_USE_SYSTEM_WXWIDGETS)
    find_package(wxWidgets 3.2 CONFIG QUIET COMPONENTS core base)
    if(NOT TARGET wx::core AND NOT TARGET wxWidgets::core)
        find_package(wxWidgets 3.2 MODULE QUIET COMPONENTS core base)
        if(wxWidgets_FOUND AND WIN32)
            list(APPEND distribution_runtime_dirs ${wxWidgets_LIBRARY_DIRS} ${wxWidgets_LIB_DIR})
        endif()
    endif()
endif()

if(TARGET wx::core)
    target_link_libraries(app PRIVATE wx::core)
elseif(TARGET wxWidgets::core)
    target_link_libraries(app PRIVATE wxWidgets::core)
elseif(TARGET wxWidgets::wxWidgets)
    target_link_libraries(app PRIVATE wxWidgets::wxWidgets)
elseif(PROJECT_USE_SYSTEM_WXWIDGETS AND wxWidgets_FOUND)
    include("${wxWidgets_USE_FILE}")
    target_link_libraries(app PRIVATE ${wxWidgets_LIBRARIES})
else()
    include(FetchContent)
    set(wxBUILD_SHARED OFF CACHE BOOL "Build wxWidgets statically")
    set(wxBUILD_SAMPLES OFF CACHE BOOL "Build wxWidgets samples")
    set(wxBUILD_TESTS OFF CACHE BOOL "Build wxWidgets tests")
    set(wxBUILD_DEMOS OFF CACHE BOOL "Build wxWidgets demos")
    set(wxUSE_WEBVIEW OFF CACHE BOOL "Build the wxWidgets web view")
    if(WIN32)
        set(wx_archive wxWidgets-3.2.10.zip)
        set(wx_hash 090caf1478b8384ea471fe9d718e48036bf6f501c2247ba353ec0e63ac44f38a)
    else()
        set(wx_archive wxWidgets-3.2.10.tar.bz2)
        set(wx_hash d66e929569947a4a5920699539089a9bda83a93e5f4917fb313a61f0c344b896)
    endif()
    FetchContent_Declare(wxwidgets
        URL "https://github.com/wxWidgets/wxWidgets/releases/download/v3.2.10/${wx_archive}"
        URL_HASH "SHA256=${wx_hash}"
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    )
    FetchContent_MakeAvailable(wxwidgets)
    set_property(DIRECTORY "${wxwidgets_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
    target_link_libraries(app PRIVATE wx::core)
endif()
